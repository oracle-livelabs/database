SELECT 
    spl.PLAN_NAME AS "Plan Name",
    spl.CATEGORY AS "Category",
    SUM(sol.LINE_TOTAL) AS "Summed Line Total",
    SUM(sol.CONNECTION_COUNT) AS "Summed Connection Count"
FROM 
    "LLUSER"."SERVICE_PLANS" spl
JOIN 
    "LLUSER"."SERVICE_ORDER_LINES" sol ON spl.PLAN_ID = sol.PLAN_ID
JOIN 
    "LLUSER"."SERVICE_ORDERS" so ON sol.ORDER_ID = so.ORDER_ID
WHERE 
    so.ORDER_STATUS IN ('confirmed', 'active', 'completed')
GROUP BY 
    spl.PLAN_NAME, 
    spl.CATEGORY
ORDER BY 
    "Summed Line Total" DESC
